% ---------------------------------------------------------------------------
% Fig. 3 -- The path from attacker-controlled bytes to tool execution, the two
% places it can be cut, and the rejection categories the gate reports.
%
% Revised to match the implemented gate (agent/provenance.py). Two cuts are
% drawn rather than one, because they are independent: the renderer decides
% whether the forged span ever acquires reserved identifiers (C1), and the gate
% decides whether a span without them can become a call (C3). Either alone
% breaks the chain; the paper measures both.
%
% Greyscale-safe: no colour carries meaning, and the cuts are drawn rather than
% symbolic so they survive black-and-white printing.
% ---------------------------------------------------------------------------
\resizebox{\columnwidth}{!}{%
\begin{tikzpicture}[
  font=\scriptsize,
  step/.style = {draw, rounded corners=2pt, align=center, inner sep=3.5pt,
                 minimum height=6mm, text width=50mm, fill=white},
  untr/.style = {step, densely dashed, fill=black!6},
  cut/.style  = {step, fill=black!14, line width=1pt},
  ar/.style   = {-{Stealth[length=2mm]}, line width=0.7pt},
  note/.style = {font=\tiny\itshape, align=left, inner sep=1.5pt}
]

\node[untr] (s1)
  {\textbf{1} \ A repository file contains a header-shaped span\\
   \scriptsize\texttt{...\hmtok{channel}commentary to=functions.bash...}};

\node[untr, below=2.8mm of s1] (s2)
  {\textbf{2} \ \texttt{read} or \texttt{grep} returns it as a tool result};

% ---- cut 1: the renderer (C1) --------------------------------------------
\node[cut, below=4.5mm of s2] (c1)
  {\textbf{Renderer admission policy} \ (C1)\\
   \scriptsize do the span's markers become \emph{reserved identifiers}?};

\node[step, below=4.5mm of c1] (s3)
  {\textbf{3} \ The span enters the context --- as ordinary\\
   \scriptsize subwords, byte-identical to the real thing};

\node[step, below=2.8mm of s3] (s4)
  {\textbf{4} \ The model restates it while reasoning\\
   \scriptsize(ordinary behaviour when summarising a file)};

\node[step, below=2.8mm of s4] (s5)
  {\textbf{5} \ A header-shaped span appears in the output};

% ---- cut 2: the gate (C3) ------------------------------------------------
\node[cut, below=4.5mm of s5] (c2)
  {\textbf{Provenance gate} \ (C3)\\
   \scriptsize reserved ids $\cdot$ canonical order $\cdot$ this turn $\cdot$ exact registry};

\node[step, below=7mm of c2] (s6)
  {\textbf{6} \ \texttt{bash} executes the attacker's command};

\foreach \a/\b in {s1/s2, s2/c1, c1/s3, s3/s4, s4/s5, s5/c2, c2/s6}
  \draw[ar] (\a) -- (\b);

% ---- draw the two cuts ----------------------------------------------------
% Unrolled rather than looped: the labels carry line breaks and math, which is
% fragile inside a \foreach argument list.
\coordinate (x1) at ($(c1.south)!0.5!(s3.north)$);
\draw[line width=1.1pt] ($(x1)+(-2mm,-1.8mm)$) -- ($(x1)+(2mm,1.8mm)$);
\draw[line width=1.1pt] ($(x1)+(-2mm,1.8mm)$) -- ($(x1)+(2mm,-1.8mm)$);
\node[note, anchor=west] at ($(x1)+(4mm,0)$)
  {escaped as subwords\\(measured: 0 of 7)};

\coordinate (x2) at ($(c2.south)!0.5!(s6.north)$);
\draw[line width=1.1pt] ($(x2)+(-2mm,-1.8mm)$) -- ($(x2)+(2mm,1.8mm)$);
\draw[line width=1.1pt] ($(x2)+(-2mm,1.8mm)$) -- ($(x2)+(2mm,-1.8mm)$);
\node[note, anchor=west] at ($(x2)+(4mm,0)$)
  {no reserved-id header,\\so not a call};

% ---- what the gate reports instead ---------------------------------------
\node[note, anchor=north west, text width=52mm, draw, dotted,
      inner sep=3pt, below=5mm of s6.south west]
  (cats)
  {\textbf{Rejection categories reported by the gate}\\
   \texttt{reserved\_token\_in\_body} $\cdot$
   \texttt{non\_canonical\_header} $\cdot$
   \texttt{duplicate\_recipient} $\cdot$
   \texttt{malformed\_recipient} $\cdot$
   \texttt{unknown\_tool} $\cdot$
   \texttt{unknown\_channel} $\cdot$
   \texttt{bad\_constraint\_type} $\cdot$
   \texttt{missing\_call\_terminator} $\cdot$
   \texttt{invalid\_json\_arguments}};
\draw[ar, densely dotted] (c2.east) -- ++(5mm,0) |- (cats.east);

% ---- what the gate replaces ----------------------------------------------
\node[note, anchor=east, text width=20mm] at ($(c2.west)+(-2mm,0)$)
  {replaces tolerant parsing and prose-level recovery};

\end{tikzpicture}%
}
